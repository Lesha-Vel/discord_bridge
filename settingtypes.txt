[Bridge]
# if this setting isn't set to localhost then server has to be ran with --allow-remote argument
discord_bridge.host (Server address) string 127.0.0.1
# if not set to default then server has to be ran with --port argument matching this setting
discord_bridge.port (Server port) int 9692 1 65535
discord_bridge.setup_token (Discord bot token) string
# comma separated list of channel ids and corresponding name to be used for that channel in format 'id[=name],id[=name]...', e.g. "123 = abc, 456", if name is not present then it defaults to "Discord"
discord_bridge.setup_channel_ids (Discord channel id list) string
# e.g. if it's set to ! and someone types !cmd to Discord chat then it will be threated as cmd command for example
discord_bridge.setup_command_prefix (Prefix for Discord prefix commands) string !
# time in seconds before player logged in through Discord will logout, 0 to disable
discord_bridge.auto_logout_timeout (Auto logout timeout) int 604800 0
# filename for login logging, empty disables this, this is relative to the world directory
discord_bridge.login_file (Login log filename) string discord_logins.txt
# see https://www.lua.org/pil/22.1.html
discord_bridge.date (Date format string) string %m/%d/%Y %I:%M%p

discord_bridge.persistent_logins (Save logins between server restarts) bool true
# this is useful to prevent flooding Discord API

# clears invites from messages sent from luanti
discord_bridge.clean_invites (Clean Discord invites) bool true
discord_bridge.setup_use_nicknames (Use Discord nicknames instead of Discord usernames) bool true

[Texts]
# this has to have 3 of %s format specifiers, first is user, second is server, third is the message itself
discord_bridge.chat_message_format (Format of the messages from Discord) string <%s@%s> %s
# this is used for messages coming from luanti, this has to have a %s format specifier for username
discord_bridge.name_wrapper (luanti message prefix, 2 spaces at the end are important) string <**%s**>  
discord_bridge.startup_text (Text of notification on server startup) string *** Server started!
discord_bridge.shutdown_text (Text of notification on server shut-down) string *** Server shutting down...
# this has to have a %s format specifier for playername
discord_bridge.join_text (Text of notification on player join) string \\*\\*\\* **%s** joined the game
# this has to have 2 of %s format specifiers, first for player name and second is for last login
discord_bridge.last_login_text (Text of notification on player join including last login) string \\*\\*\\* **%s** joined the game. Last login: %s
# this has to have a %s format specifier for playername
discord_bridge.leave_text (Text of notification on player leave) string \\*\\*\\* **%s** left the game
# this has to have a %s format specifier for playername
discord_bridge.welcome_text (Text of notification on player welcome) string \\*\\*\\* **%s** joined the game for the first time. Welcome!
# this has to have a %s format specifier for playername
discord_bridge.death_text (Text of notification on player death) string \\*\\*\\* **%s** died
